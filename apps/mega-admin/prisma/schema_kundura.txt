// Ana Ürün (Master Product = Fiziksel Gerçeklik)
model Product {
  id          String  @id @default(cuid())
  FirmId     String
  Firm       Firm   @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  name        String
  slug        String
  description String  @db.Text
  shortDesc   String?

  // ── Fiyatlandırma ──
  price          Float
  compareAtPrice Float? // Üstü çizili eski fiyat
  costPrice      Float? // Maliyet (müşteri görmez)

  // ── Sınıflandırma ──
  category ProductCategory @default(OTHER)
  status   ProductStatus   @default(ACTIVE)
  vendor   String?         @default("Davut Kundura Atölyesi")
  tags     String[]

  // ── Medya ──
  imageUrls  String[] @db.Text
  videoUrl   String? // Ürün tanıtım videosu
  modelUrl3D String? // AR için .glb / .usdz dosyası

  // ── Stok & Lojistik ──
  sku            String?
  barcode        String?
  weight         Float? // gram
  lowStockAlert  Int     @default(5)
  trackInventory Boolean @default(true)
  allowBackorder Boolean @default(false)

  // ── SEO ──
  seoTitle    String?
  seoDesc     String?
  seoKeywords String[]

  // ── İlişkiler ──
  channelVisibility ChannelVisibility?
  variants          ProductVariant[]
  options           ProductOption[] // "Renk", "Numara" gibi seçenek grupları
  orderItems        OrderItem[]
  inventory         InventoryTransaction[]
  reviews           Review[]
  wishlistItems     WishlistItem[]
  productionOrders  ProductionOrder[]
  bundleItems       ProductBundleItem[]    @relation("BundleComponents") // Bu ürün bir paketin parçası

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@unique([FirmId, slug])
  @@unique([FirmId, sku])
}

// Sanal Paket (Bundle): Birden fazla ürünü tek paket olarak satar,
// satıldığında her bileşenin stoğunu otomatik düşer.
model ProductBundle {
  id          String              @id @default(cuid())
  FirmId     String
  Firm       Firm               @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  name        String // "Profesyonel Bakım Seti"
  description String?             @db.Text
  price       Float // Paketin toplam satış fiyatı
  imageUrl    String?
  isActive    Boolean             @default(true)
  items       ProductBundleItem[]
  createdAt   DateTime            @default(now())
  updatedAt   DateTime            @updatedAt
}

model ProductBundleItem {
  id        String        @id @default(cuid())
  FirmId   String
  Firm     Firm         @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  bundleId  String
  bundle    ProductBundle @relation(fields: [bundleId], references: [id], onDelete: Cascade)
  productId String
  product   Product       @relation("BundleComponents", fields: [productId], references: [id])
  variantId String? // Belirli bir varyant kilitlenebilir (örn: sadece Siyah)
  quantity  Int           @default(1) // Pakette kaç adet var?

  @@unique([FirmId, bundleId, productId, variantId])
}

// Ürün seçenek grubu (Örn: "Renk", "Numara", "Materyal")
model ProductOption {
  id        String               @id @default(cuid())
  FirmId   String
  Firm     Firm                @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  productId String
  product   Product              @relation(fields: [productId], references: [id], onDelete: Cascade)
  name      String // "Renk", "Numara"
  position  Int                  @default(0) // Gösterim sırası
  values    ProductOptionValue[]
}

// Her seçeneğin alabileceği değerler
model ProductOptionValue {
  id               String        @id @default(cuid())
  FirmId          String
  Firm            Firm         @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  optionId         String
  option           ProductOption @relation(fields: [optionId], references: [id], onDelete: Cascade)
  value            String // "Siyah", "42", "Süet"
  // Koşullu görünürlük: Sadece belirli bir üst değer seçilince görünür
  // Örn: Sadece "Siyah" renk seçilince "Süet" materyali aç
  dependsOnValueId String?
  position         Int           @default(0)
}

// Ürün Varyantı (Her kombinasyon = 1 varyant, kendi SKU ve stokuyla)
model ProductVariant {
  id        String  @id @default(cuid())

  productId String
  product   Product @relation(fields: [productId], references: [id], onDelete: Cascade)

  // ── Tanımlama ──
  name     String // "42 Numara - Siyah - Hakiki Deri"
  size     String?
  color    String?
  material String? // Yeni: Materyal varyasyonu

  // ── Fiyat & Maliyet ──

  costPrice Float? // Varyanta özel maliyet

  // ── Stok & Kimlik ──
  sku      String? @unique
  barcode  String? // Her varyantın kendi barkodu
  stock    Int     @default(0)
  imageUrl String?
  isActive Boolean @default(true)

  // ── İlişkiler ──
  orderItems      OrderItem[]
  warehouseStocks WarehouseStock[]
  stockLedger     StockLedger[]
}

enum OrderStatus {
  PENDING // Bekliyor
  PROCESSING // Hazırlanıyor
  PARTIAL // Parçalı Gönderim
  SHIPPED // Kargoya verildi
  DELIVERED // Teslim edildi
  CANCELLED // İptal edildi
  REFUNDED // İade edildi
}

enum PaymentStatus {
  UNPAID
  PARTIALLY_PAID // B2B parçalı ödeme
  PAID
  PARTIALLY_REFUNDED
  REFUNDED
}

enum OrderType {
  B2C // Perakende
  B2B // Toptan / Bayi
}

model Order {
  id          String    @id @default(cuid())
  FirmId     String
  Firm       Firm     @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  orderNumber String    @unique // ORD-2024-0001
  orderType   OrderType @default(B2C)

  userId           String
  user             User     @relation(fields: [userId], references: [id])
  addressId        String?
  address          Address? @relation(fields: [addressId], references: [id])
  billingAddressId String?

  status        OrderStatus   @default(PENDING)
  paymentStatus PaymentStatus @default(UNPAID)

  // Finansal Veriler
  subtotal       Float
  taxAmount      Float @default(0)
  shippingCost   Float @default(0)
  discountAmount Float @default(0)
  totalAmount    Float
  paidAmount     Float @default(0) // B2B parçalı ödeme takibi için

  // B2B Net Gün
  dueDate DateTime? // Cari açık hesap son ödeme tarihi

  // Paytr / Ödeme Altyapısı
  paytrMerchantOid String?
  paytrToken       String?

  discountCodeId String?
  discountCode   DiscountCode? @relation(fields: [discountCodeId], references: [id])

  notes      String? @db.Text // Müşteri notu
  adminNotes String? @db.Text // Admin notu

  // İlişkiler
  items          OrderItem[]
  fulfillments   Fulfillment[]
  refunds        Refund[]
  eventLogs      OrderEventLog[]
  invoice        Invoice?
  supportTickets SupportTicket[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model OrderItem {
  id      String @id @default(cuid())
  FirmId String

  product   Product         @relation(fields: [productId], references: [id])
  variantId String?

  refundedQty  Int @default(0)

  bundleId String? // Eğer bu bir bundle içindeki ürünse
}

enum StockMovementType {
  PURCHASE // Tedarikçiden satın alma girişi
  SALE // Sipariş satış çıkışı

  RETURN_OUT // Tedarikçiye iade çıkışı

// ERP-grade stok defteri: Stok miktarı doğrudan güncellenmez,
// her hareket buraya yazılır. Gerçek stok = SUM(quantity) per variant+warehouse
model StockLedger {
  id           String            @id @default(cuid())
  FirmId      String
  Firm        Firm             @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  variantId    String
  variant      ProductVariant    @relation(fields: [variantId], references: [id])
  warehouseId  String
  warehouse    Warehouse         @relation(fields: [warehouseId], references: [id])
  movementType StockMovementType
  quantity     Int // Pozitif: giriş, Negatif: çıkış
  unitCost     Float? // Hareket anındaki birim maliyet (FIFO için)
  reference    String? // Sipariş no, tedarik no, vb.
  notes        String?
  performedBy  String? // Admin user ID veya "system"
  createdAt    DateTime          @default(now())

  @@index([variantId, warehouseId])
  @@index([createdAt])
}

model ShippingProvider {
  id       String               @id @default(cuid())
  FirmId  String
  Firm    Firm                @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  name     String // "Yurtiçi Kargo", "DHL Express"
  slug     String               @unique // "yurtici", "dhl"
  type     ShippingProviderType
  logoUrl  String?
  isActive Boolean              @default(true)

  // API Entegrasyon Bilgileri (şifrelenmiş)
  apiBaseUrl String?
  apiKey     String? // Şifreli
  apiSecret  String? // Şifreli
  webhookUrl String? // Gelen tracking eventi için

  dispatches Dispatch[]
  zones      ShippingZone[]

  createdAt DateTime @default(now())
}

model ShippingZone {
  id         String           @id @default(cuid())
  FirmId    String
  Firm      Firm            @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  providerId String
  provider   ShippingProvider @relation(fields: [providerId], references: [id])
  name       String // "Türkiye İçi", "Avrupa", "ABD"
  countries  String[] // ["TR"], ["DE","FR","NL"]

  // Desi (Hacimsel Ağırlık) Bazlı Fiyatlandırma
  baseFee     Float  @default(0) // Taban ücret
  perDesiRate Float  @default(0) // Her desi için ek ücret (₺ veya €)
  currency    String @default("TRY")
  freeAbove   Float? // Bu tutarın üzerinde kargo bedava

  @@index([providerId])
}

// Depodaki standart kutu tipleri — Desi hesabı için

  FirmId       String
  Firm         Firm  @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  name          String // "Tekli Ayakkabı Kutusu", "Çizme Kutusu", "B2B Palet"
  widthCm       Float

  // Bu alan Prisma'da computed olmadığı için API'de hesaplanır
  dispatches Dispatch[]

// Ana Sevkiyat/Dispatch kaydı — Her kargo etiketine karşılık gelir
enum DispatchStatus {
  DRAFT // Henüz API'ye gönderilmedi
  LABEL_READY // Etiket hazır, yazdırılmayı bekliyor
  PRINTED // Etiketi yazdırıldı
  HANDED_OFF // Kargo firmasına teslim edildi
  IN_TRANSIT // Yolda
  DELIVERED // Teslim edildi
  FAILED // Teslim edilemedi
  RETURNED // İade kargoya verildi
}

model Dispatch {
  id              String           @id @default(cuid())
  FirmId         String
  Firm           Firm            @relation(fields: [FirmId], references: [id], onDelete: Cascade)
  fulfillmentId   String           @unique
  fulfillment     Fulfillment      @relation(fields: [fulfillmentId], references: [id])
  providerId      String
  provider        ShippingProvider @relation(fields: [providerId], references: [id])
  packagePresetId String?
  packagePreset   PackagePreset?   @relation(fields: [packagePresetId], references: [id])

  status DispatchStatus @default(DRAFT)

  // Kargo Detayları
  trackingNo  String? @unique
  trackingUrl String?
  zplLabel    String? @db.Text // Zebra termal yazıcı için ZPL verisi
  pdfLabelUrl String? // PDF etiket linki

  // Boyut & Ağırlık
  grossWeightKg    Float?
  desiWeight       Float? // Hesaplanan desi ağırlığı
  chargeableWeight Float? // max(gerçek, desi)

  // Fatura & İrsaliye
  waybillNo    String? // E-İrsaliye numarası
  shippingCost Float? // Kargo firmasından dönen gerçek maliyet

  // Tersine lojistik için iade kodu
  returnBarcode String?

  trackingEvents TrackingEvent[]

  createdAt DateTime @default(now())
